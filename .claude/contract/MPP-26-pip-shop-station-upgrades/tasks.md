```sql
-- MPP-26: Pip's station levels (bed, fridge, cooker, trophy cabinet), per person, plus the ledger
-- source for spending sprinkles on them.
--
-- pet_station_levels - one row per station a person has upgraded; level 2..4. A missing row is
-- level 1, so every user (existing and new) starts at 1 with no backfill (AC 1). Written only by
-- StationService.upgrade under the per-person wallet lock; nothing lowers a level or deletes a row,
-- so upgrades are never lost (AC 4).
--
-- pet_sprinkles.chk_pet_sprinkles_source is re-declared as a superset of MPP-25's list, adding
-- STATION_UPGRADE: a spend is one negative row, source_id = station code * 10 + new level
-- (bed 1, fridge 2, cooker 3, cabinet 4 -> 12..44), so uk_pet_sprinkles also lets each person buy
-- each (station, level) once.
--
-- Per person (MPP-33): user_id is the person upgrading their own room, never the meal-plan owner.
--
-- Lossless: one new, empty table, and a CHECK widened to a superset (every existing row still
-- passes). Local evidence 2026-10-14: 13 users (2 linked), no station table yet.
--
-- Idempotent (MPP-15 go-live file): CREATE TABLE IF NOT EXISTS, and the CHECK is dropped only if
-- present before being re-added, so re-running the file is safe.
--
-- MUST run after MPP-25's 2026-10-14-mpp25-pet-sprinkles.sql (go-live row 7) and before the
-- backend that maps PetStationLevel starts (ddl-auto: validate). MPP-15 is local-first: applied to
-- the local foodbytes-db now; row 8 of .claude/contract/MPP-15-epic-breakdown/railway-migrations.md
-- for Railway at merge time.

CREATE TABLE IF NOT EXISTS pet_station_levels (
  id          BIGINT      NOT NULL AUTO_INCREMENT,
  user_id     BIGINT      NOT NULL,
  station     VARCHAR(16) NOT NULL,
  level       INT         NOT NULL,
  upgraded_at TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_pet_station_levels (user_id, station),
  CONSTRAINT chk_pet_station_levels_station CHECK (station IN ('BED','FRIDGE','COOKER','CABINET')),
  CONSTRAINT chk_pet_station_levels_level CHECK (level BETWEEN 2 AND 4),
  CONSTRAINT fk_pet_station_levels_user FOREIGN KEY (user_id)
    REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

SET @has_chk := (SELECT COUNT(*) FROM information_schema.CHECK_CONSTRAINTS
                 WHERE CONSTRAINT_SCHEMA = DATABASE() AND CONSTRAINT_NAME = 'chk_pet_sprinkles_source');
SET @drop_chk := IF(@has_chk > 0, 'ALTER TABLE pet_sprinkles DROP CHECK chk_pet_sprinkles_source', 'DO 0');
PREPARE drop_chk FROM @drop_chk;
EXECUTE drop_chk;
DEALLOCATE PREPARE drop_chk;

ALTER TABLE pet_sprinkles ADD CONSTRAINT chk_pet_sprinkles_source CHECK (source_type IN
  ('MEAL_EATEN','MEAL_ON_TIME','FULL_WEEK','HOMEMADE_EXTRA','STATION_UPGRADE'));
```

```java
package com.foodbytes.model;

import java.util.Arrays;
import java.util.Locale;

/**
 * MPP-26: Pip's four upgradable stations. code feeds the ledger source id (StationPolicy.sourceId);
 * key is the JSON / client key (constants/petRoom.js ROOM_STATIONS); label is for messages.
 */
public enum RoomStation {
    BED(1, "bed", "bed"),
    FRIDGE(2, "fridge", "fridge"),
    COOKER(3, "cooker", "cooker"),
    CABINET(4, "cabinet", "trophy cabinet");

    private final int code;
    private final String key;
    private final String label;

    RoomStation(int code, String key, String label) {
        this.code = code;
        this.key = key;
        this.label = label;
    }

    public int code() { return code; }
    public String key() { return key; }
    public String label() { return label; }

    /** The path variable, case-insensitive ("bed", "BED"). */
    public static RoomStation fromPath(String value) {
        String wanted = value == null ? "" : value.trim().toLowerCase(Locale.ROOT);
        return Arrays.stream(values())
                .filter(station -> station.key.equals(wanted))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Unknown station: " + value));
    }
}
```

- [x] **Step 3: Implement** (append to `SprinkleService`; imports `org.springframework.dao.DataIntegrityViolationException`; extend the class Javadoc with "MPP-26: lockWallet + spend - the one locked debit path every purchase uses.")

```java
    /**
     * MPP-26: SELECT … FOR UPDATE on the caller's users row. Every purchase takes it first, so one
     * person's spends run one at a time and the balance can't be overdrawn by two taps or devices.
     * MANDATORY: the lock lives as long as the caller's transaction.
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public void lockWallet(Long userId) {
        userRepository.lockById(userId)
                .orElseThrow(() -> new IllegalArgumentException("User not found: " + userId));
    }

    /**
     * MPP-26: one spend - checks the balance and writes one negative ledger row. The caller holds
     * lockWallet. Earnings never lock and only add, so the check can't go stale before commit.
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public void spend(Long userId, SprinkleSource source, long sourceId, int cost) {
        long balance = petSprinkleRepository.sumAmountByUserId(userId);
        if (balance < cost) {
            throw new PetShopException(PetShopException.Reason.NOT_ENOUGH_SPRINKLES,
                    "Not enough sprinkles: you have " + balance + ", this costs " + cost + ".");
        }
        try {
            petSprinkleRepository.insertDebit(userId, source.name(), sourceId, -cost);
        } catch (DataIntegrityViolationException e) {
            throw new PetShopException(PetShopException.Reason.ALREADY_OWNED, "You already have that.");
        }
    }
```

```java
package com.foodbytes.service;

import com.foodbytes.dto.PetRoomDTO;
import com.foodbytes.dto.RoomWriteDTO;
import com.foodbytes.model.PetStationLevel;
import com.foodbytes.model.RoomStation;
import com.foodbytes.repository.PetStationLevelRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.EnumMap;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

/**
 * MPP-26: the caller's own room as the client sees it (per-person Pip, MPP-33 - never the meal-plan
 * owner's). Built from repositories only, so the services that change the room (StationService,
 * MPP-27's DecorService) depend on this and never on each other.
 */
@Service
@RequiredArgsConstructor
public class PetRoomService {

    private final PetStationLevelRepository levelRepository;
    private final SprinkleService sprinkleService;

    @Transactional(readOnly = true)
    public PetRoomDTO room(Long userId) {
        Map<RoomStation, Integer> levels = levels(userId);
        Map<String, Integer> stations = new LinkedHashMap<>();
        for (RoomStation station : RoomStation.values()) {
            stations.put(station.key(), levels.get(station));
        }
        return new PetRoomDTO(stations, new TreeMap<>(StationPolicy.UPGRADE_COSTS));
    }

    /** The response to every room write: the balance after it, and the whole room. */
    @Transactional(readOnly = true)
    public RoomWriteDTO write(Long userId) {
        return new RoomWriteDTO(sprinkleService.balance(userId), room(userId));
    }

    /** Every station's level; a station with no row is at level 1. */
    public Map<RoomStation, Integer> levels(Long userId) {
        return levelsOf(levelRepository.findByUserId(userId));
    }

    static Map<RoomStation, Integer> levelsOf(List<PetStationLevel> rows) {
        Map<RoomStation, Integer> levels = new EnumMap<>(RoomStation.class);
        for (RoomStation station : RoomStation.values()) {
            levels.put(station, StationPolicy.MIN_LEVEL);
        }
        rows.forEach(row -> levels.put(row.getStation(), row.getLevel()));
        return levels;
    }
}
```

```java
package com.foodbytes.service;

import com.foodbytes.dto.RoomWriteDTO;
import com.foodbytes.model.RoomStation;
import com.foodbytes.model.SprinkleSource;
import com.foodbytes.repository.PetStationLevelRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * MPP-26 AC 2: raise one of the caller's own stations by one level, paying from their own
 * sprinkles. Lock → level check → debit → level up, in one transaction: any failure rolls back
 * both writes, so nobody pays without the level or levels up without paying.
 */
@Service
@RequiredArgsConstructor
public class StationService {

    private final PetStationLevelRepository levelRepository;
    private final SprinkleService sprinkleService;
    private final PetRoomService petRoomService;

    @Transactional
    public RoomWriteDTO upgrade(Long userId, RoomStation station) {
        sprinkleService.lockWallet(userId);
        int level = PetRoomService.levelsOf(levelRepository.findByUserId(userId)).get(station);
        if (StationPolicy.isMaxed(level)) {
            throw new PetShopException(PetShopException.Reason.MAXED,
                    "Your " + station.label() + " is already at the top level.");
        }
        int next = level + 1;
        sprinkleService.spend(userId, SprinkleSource.STATION_UPGRADE, StationPolicy.sourceId(station, next), StationPolicy.costFor(next));
        levelRepository.upsert(userId, station.name(), next);
        return petRoomService.write(userId);
    }
}
```

```java
import com.foodbytes.model.RoomStation;
```

```java
import lombok.RequiredArgsConstructor;

import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;

/**
 * MPP-26: upgrade one of the caller's own stations (see StationService). The response is the
 * RoomWriteDTO the client swaps in; reads ride on GET /api/pet (PetDTO.room).
 */
@RestController
@RequestMapping("/api/pet/stations")
@RequiredArgsConstructor
public class PetStationController {

    private final StationService stationService;

    @PostMapping("/{station}/upgrade")
    public ResponseEntity<?> upgrade(
            @AuthenticationPrincipal UserPrincipal userPrincipal,
            @PathVariable String station) {
        if (userPrincipal == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).body(Map.of("error", "Not authenticated"));
        }
        RoomStation resolved;
        try {
            resolved = RoomStation.fromPath(station);
        } catch (IllegalArgumentException e) {
            return ResponseEntity.badRequest().body(Map.of("error", e.getMessage()));
        }
        try {
            return ResponseEntity.ok(stationService.upgrade(userPrincipal.getId(), resolved));
        } catch (PetShopException e) {
            return ResponseEntity.status(HttpStatus.CONFLICT).body(Map.of("error", e.getMessage()));
        }
    }
```

```js
import { useEffect, useState } from 'react'
import petService from '../services/petService'
import { AWARD_BUBBLE_MS } from '../constants/petAwards'
import { SHOP_CONFIRM_MS, SHOP_STATIONS, SHOP_UPGRADE_ERROR, SHOP_UPGRADE_LINES, STATION_LEVEL_NAMES } from '../constants/petShop'

/**
 * MPP-26: the Shop sheet's state. A first tap arms a station (SHOP_CONFIRM_MS); a second tap
 * upgrades it. Purchases are NOT optimistic — spending can't be undone, so the room changes only
 * when the server says so (applyRoomWrite). A 409 shows the server's message and refetches the pet
 * so the sheet shows the true balance and levels. Writes are never aborted.
 *
 * @param {{ applyRoomWrite: (w: { sprinkles: number, room: Object }) => void, refreshPet: () => void }} input
 * @returns {{ armed: string|null, busy: string|null, status: string|null, error: string|null,
 *   announcement: string|null, arm: (station: string) => void, upgrade: (station: string) => void, reset: () => void }}
 */
export function usePetShop({ applyRoomWrite, refreshPet }) {
  const [armed, setArmed] = useState(null)
  const [busy, setBusy] = useState(null)
  const [status, setStatus] = useState(null)
  const [error, setError] = useState(null)
  const [announcement, setAnnouncement] = useState(null)

  useEffect(() => {
    if (!armed) return undefined
    const timerId = setTimeout(() => setArmed(null), SHOP_CONFIRM_MS)
    return () => clearTimeout(timerId)
  }, [armed])

  useEffect(() => {
    if (!announcement) return undefined
    const timerId = setTimeout(() => setAnnouncement(null), AWARD_BUBBLE_MS)
    return () => clearTimeout(timerId)
  }, [announcement])

  const arm = (station) => {
    setArmed(station)
    setStatus(null)
    setError(null)
  }

  const upgrade = (station) => {
    if (busy) return
    setArmed(null)
    setBusy(station)
    setStatus(null)
    setError(null)
    petService.upgradeStation(station)
      .then(result => {
        applyRoomWrite(result)
        setStatus(upgradedLine(station, result.room?.stations?.[station]))
        setAnnouncement(SHOP_UPGRADE_LINES[station] ?? null)
      })
      .catch(err => {
        setError(err.response?.data?.error || SHOP_UPGRADE_ERROR)
        if (err.response?.status === 409) refreshPet()
      })
      .finally(() => setBusy(null))
  }

  // The sheet closed: forget the armed button and the last message.
  const reset = () => {
    setArmed(null)
    setStatus(null)
    setError(null)
  }

  return { armed, busy, status, error, announcement, arm, upgrade, reset }
}

function upgradedLine(station, level) {
  const label = SHOP_STATIONS.find(s => s.key === station)?.label ?? station
  const name = STATION_LEVEL_NAMES[station]?.[level - 1]
  return name ? `${label} upgraded to level ${level}: ${name}!` : `${label} upgraded!`
}

export default usePetShop
```
