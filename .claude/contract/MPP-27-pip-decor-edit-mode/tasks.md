```sql
-- MPP-27: Pip's room decor, per person - what each person has bought (kept forever) and what sits
-- on each of the room's fixed spots, plus the ledger source for paying for it.
--
-- pet_decor_owned - one row per decor item or style a person has bought. Never deleted by the app
-- (AC 2: owned forever). The free default styles (wall-lavender, floor-lilac) are owned implicitly
-- and never stored. Written only by DecorService.buy under the per-person wallet lock.
--
-- pet_room_placements - one row per filled spot: LEFT_WALL / RIGHT_WALL / FLOOR hold a decor item,
-- WALL_STYLE / FLOOR_STYLE hold the chosen style. A missing row is an empty spot, or the default
-- style, so every user (existing and new) starts with today's room and no backfill. item_key is not
-- CHECK-constrained: the catalogue is data in DecorCatalogue.java (the pet_wardrobe_unlocks precedent).
--
-- pet_sprinkles.chk_pet_sprinkles_source is re-declared as a superset of MPP-26's list, adding
-- DECOR_PURCHASE: a purchase is one negative row with source_id = the pet_decor_owned.id inserted in
-- the same transaction, so each debit points at what it bought.
--
-- Per person (MPP-33): user_id is the person decorating their own room, never the meal-plan owner.
--
-- Lossless: two new, empty tables, and a CHECK widened to a superset (every existing row still
-- passes). Local evidence 2026-10-14: no decor or room tables exist; 13 users (2 linked).
--
-- Idempotent (MPP-15 go-live file): CREATE TABLE IF NOT EXISTS, and the CHECK is dropped only if
-- present before being re-added, so re-running the file is safe.
--
-- MUST run after MPP-25 (go-live row 7, creates pet_sprinkles) and MPP-26 (row 8), and before the
-- backend that maps PetDecorOwned / PetRoomPlacement starts (ddl-auto: validate). MPP-15 is
-- local-first: applied to the local foodbytes-db now; row 9 of
-- .claude/contract/MPP-15-epic-breakdown/railway-migrations.md for Railway at merge time.

CREATE TABLE IF NOT EXISTS pet_decor_owned (
  id           BIGINT      NOT NULL AUTO_INCREMENT,
  user_id      BIGINT      NOT NULL,
  item_key     VARCHAR(32) NOT NULL,
  purchased_at TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_pet_decor_owned (user_id, item_key),
  CONSTRAINT fk_pet_decor_owned_user FOREIGN KEY (user_id)
    REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS pet_room_placements (
  id        BIGINT      NOT NULL AUTO_INCREMENT,
  user_id   BIGINT      NOT NULL,
  spot      VARCHAR(16) NOT NULL,
  item_key  VARCHAR(32) NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uk_pet_room_placements (user_id, spot),
  CONSTRAINT chk_pet_room_placements_spot CHECK (spot IN ('LEFT_WALL','RIGHT_WALL','FLOOR','WALL_STYLE','FLOOR_STYLE')),
  CONSTRAINT fk_pet_room_placements_user FOREIGN KEY (user_id)
    REFERENCES users (id) ON DELETE CASCADE
) ENGINE=InnoDB;

SET @has_chk := (SELECT COUNT(*) FROM information_schema.CHECK_CONSTRAINTS
                 WHERE CONSTRAINT_SCHEMA = DATABASE() AND CONSTRAINT_NAME = 'chk_pet_sprinkles_source');
SET @drop_chk := IF(@has_chk > 0, 'ALTER TABLE pet_sprinkles DROP CHECK chk_pet_sprinkles_source', 'DO 0');
PREPARE drop_chk FROM @drop_chk;
EXECUTE drop_chk;
DEALLOCATE PREPARE drop_chk;

ALTER TABLE pet_sprinkles ADD CONSTRAINT chk_pet_sprinkles_source CHECK (source_type IN
  ('MEAL_EATEN','MEAL_ON_TIME','FULL_WEEK','HOMEMADE_EXTRA','STATION_UPGRADE','DECOR_PURCHASE'));
```

```java
package com.foodbytes.model;

import java.util.Arrays;
import java.util.Locale;

/**
 * MPP-27: the room's three fixed decor spots plus its two finishes. Stored by name in
 * pet_room_placements.spot (chk_pet_room_placements_spot). key is the JSON / client key
 * (constants/petDecor.js ROOM_SPOTS); label is for messages. The style "spots" hold the chosen wall
 * and floor style, so one placement table covers both; isStyle() keeps them apart in code.
 */
public enum RoomSpot {
    LEFT_WALL("leftWall", "left wall", false),
    RIGHT_WALL("rightWall", "right wall", false),
    FLOOR("floor", "floor", false),
    WALL_STYLE("wallStyle", "walls", true),
    FLOOR_STYLE("floorStyle", "floor", true);

    private final String key;
    private final String label;
    private final boolean style;

    RoomSpot(String key, String label, boolean style) {
        this.key = key;
        this.label = label;
        this.style = style;
    }

    public String key() { return key; }
    public String label() { return label; }
    public boolean isStyle() { return style; }

    /** The request's spot: the JSON key ("leftWall") or the enum name ("LEFT_WALL", any case). */
    public static RoomSpot fromKey(String value) {
        String wanted = value == null ? "" : value.trim();
        return Arrays.stream(values())
                .filter(spot -> spot.key.equals(wanted) || spot.name().equals(wanted.toUpperCase(Locale.ROOT)))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Unknown spot: " + value));
    }
}
```

```java
                .map(Item::id)
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException(styleSpot + " has no default"));
    }
```

```java
package com.foodbytes.model;

import jakarta.persistence.*;
import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;
import java.time.LocalDateTime;

/**
 * MPP-27: one decor item or style one person has bought. Identified by (user, item), so it can be
 * bought once; saved only by DecorService.buy and never deleted by the app (AC 2: kept forever).
 * The free default styles never have a row.

@Entity
@Table(name = "pet_decor_owned", uniqueConstraints = {
    @UniqueConstraint(name = "uk_pet_decor_owned", columnNames = {"user_id", "item_key"})
})
@Data
@NoArgsConstructor
@AllArgsConstructor
public class PetDecorOwned {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    // The person decorating their own room - never the meal-plan owner (MPP-33)
    @Column(name = "user_id", nullable = false)
    private Long userId;

    // A DecorCatalogue id, e.g. "herb-shelf"
    @Column(name = "item_key", nullable = false, length = 32)
    private String itemKey;

    // DB default CURRENT_TIMESTAMP (UTC, as the JDBC URL pins serverTimezone=UTC)
    @Column(name = "purchased_at", nullable = false, insertable = false, updatable = false)
    private LocalDateTime purchasedAt;

    // Identity is the primary key — see Recipe#equals.
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof PetDecorOwned other)) return false;
        return id != null && id.equals(other.getId());
    }

    @Override
    public int hashCode() {
        return PetDecorOwned.class.hashCode();
    }
}
```

```java
package com.foodbytes.service;

import com.foodbytes.dto.RoomWriteDTO;
import com.foodbytes.model.PetDecorOwned;
import com.foodbytes.model.RoomSpot;
import com.foodbytes.model.SprinkleSource;
import com.foodbytes.repository.PetDecorOwnedRepository;
import com.foodbytes.repository.PetRoomPlacementRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

/**
 * MPP-27: buying and placing decor in the caller's own room (per-person Pip, MPP-33 - never the
 * meal-plan owner's). buy() is MPP-26's spend path plus an ownership row: lock → checks → owned row
 * → debit → auto-place, in one transaction, so a refused debit rolls the owned row back and nothing
 * is kept unpaid (AC 2). place() moves no sprinkles, so it takes no lock (AC 3).
 */
@Service
@RequiredArgsConstructor
public class DecorService {

    private final PetDecorOwnedRepository ownedRepository;
    private final PetRoomPlacementRepository placementRepository;
    private final SprinkleService sprinkleService;
    private final PetRoomService petRoomService;

    /** A bought item goes on its spot if that spot is empty; a bought style is applied at once. */
    @Transactional
    public RoomWriteDTO buy(Long userId, String itemId) {
        DecorCatalogue.Item item = known(itemId);
        sprinkleService.lockWallet(userId);
        if (item.isDefault() || ownedRepository.existsByUserIdAndItemKey(userId, item.id())) {
            throw new PetShopException(PetShopException.Reason.ALREADY_OWNED, "You already own that.");
        }
        PetDecorOwned owned = ownedRepository.save(new PetDecorOwned(null, userId, item.id(), null));
        sprinkleService.spend(userId, SprinkleSource.DECOR_PURCHASE, owned.getId(), item.price());
        RoomSpot spot = item.spot();
        if (spot.isStyle() || !placementRepository.existsByUserIdAndSpot(userId, spot)) {
            placementRepository.upsert(userId, spot.name(), item.id());
        }
        return petRoomService.write(userId);
    }

    /** Puts an owned item on its spot, or clears the spot when itemId is null/blank (a finish resets to its default). */
    @Transactional
    public RoomWriteDTO place(Long userId, RoomSpot spot, String itemId) {
        if (itemId == null || itemId.isBlank()) {
            placementRepository.deleteByUserIdAndSpot(userId, spot);
            return petRoomService.write(userId);
        }
        DecorCatalogue.Item item = known(itemId);
        if (item.spot() != spot) {
            throw new IllegalArgumentException(item.id() + " doesn't go on the " + spot.label());
        }
        if (item.isDefault()) {
            placementRepository.deleteByUserIdAndSpot(userId, spot); // the default is never stored
            return petRoomService.write(userId);
        }
        if (!ownedRepository.existsByUserIdAndItemKey(userId, item.id())) {
            throw new PetShopException(PetShopException.Reason.NOT_OWNED, "You don't own that yet.");
        }
        placementRepository.upsert(userId, spot.name(), item.id());
        return petRoomService.write(userId);
    }

    private static DecorCatalogue.Item known(String itemId) {
        return DecorCatalogue.find(itemId)
                .orElseThrow(() -> new IllegalArgumentException("Unknown item: " + itemId));
    }
}
```

```java
package com.foodbytes.controller;

import com.foodbytes.dto.RoomPlacementRequest;
import com.foodbytes.dto.RoomWriteDTO;
import com.foodbytes.model.RoomSpot;
import com.foodbytes.security.UserPrincipal;
import com.foodbytes.service.DecorService;
import com.foodbytes.service.PetShopException;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.*;
import java.util.Map;
import java.util.function.Supplier;

/**
 * MPP-27: buy decor and place it in the caller's own room (see DecorService). Both answer MPP-26's
 * RoomWriteDTO, which the client swaps in; reads ride on GET /api/pet (PetDTO.room).
 */
@RestController
@RequestMapping("/api/pet")
@RequiredArgsConstructor
public class PetDecorController {

    private final DecorService decorService;

    @PostMapping("/decor/{itemId}/buy")
    public ResponseEntity<?> buy(
            @AuthenticationPrincipal UserPrincipal userPrincipal,
            @PathVariable String itemId) {
        if (userPrincipal == null) {
            return unauthenticated();
        }
        return respond(() -> decorService.buy(userPrincipal.getId(), itemId));
    }

    @PutMapping("/room/placements")
    public ResponseEntity<?> place(
            @AuthenticationPrincipal UserPrincipal userPrincipal,
            @Valid @RequestBody RoomPlacementRequest request) {
        if (userPrincipal == null) {
            return unauthenticated();
        }
        return respond(() -> decorService.place(userPrincipal.getId(), RoomSpot.fromKey(request.getSpot()), request.getItemId()));
    }

    // PetShopException → 409 (the client's copy is stale), IllegalArgumentException → 400 (bad id / spot / kind).
    private static ResponseEntity<?> respond(Supplier<RoomWriteDTO> write) {
        try {
            return ResponseEntity.ok(write.get());
        } catch (PetShopException e) {
            return ResponseEntity.status(HttpStatus.CONFLICT).body(Map.of("error", e.getMessage()));
        } catch (IllegalArgumentException e) {
            return ResponseEntity.badRequest().body(Map.of("error", e.getMessage()));
        }
    }

    /** No @ControllerAdvice in this codebase - same {error} shape as PetWardrobeController. */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<?> handleValidation(MethodArgumentNotValidException ex) {
        String message = ex.getBindingResult().getFieldErrors().stream()
                .findFirst()
                .map(err -> err.getDefaultMessage())
                .orElse("Invalid request");
        return ResponseEntity.badRequest().body(Map.of("error", message));
    }

    private static ResponseEntity<?> unauthenticated() {
        return ResponseEntity.status(HttpStatus.UNAUTHORIZED).body(Map.of("error", "Not authenticated"));
    }
}
```

```js
/**
```

```js
 * ({ leftWall, rightWall, floor, wallStyle, floorStyle }). Pinned by room.check.mjs.

/** A known item or style's { label, spot }, or null for an id this client doesn't know. */
export function decorEntry(id) {
  return DECOR_ITEMS[id] ?? ROOM_STYLES[id] ?? null
}

/** The room after putting itemId on spot, for an optimistic edit. null clears it, and a cleared finish is its default. */
export function withPlacement(room, spot, itemId) {
  const cleared = spot === ROOM_SPOTS.WALL_STYLE ? DEFAULT_WALL_STYLE : spot === ROOM_SPOTS.FLOOR_STYLE ? DEFAULT_FLOOR_STYLE : null
  return { ...room, placements: { ...room?.placements, [spot]: itemId || cleared } }
}

/** Every known catalogue entry for a spot, in the server's order (the shop grids). */
export function catalogueFor(decor, spot) {
  return (decor ?? []).filter(item => item.kind === spot && decorEntry(item.id)?.spot === spot)
}

/** The owned, known entries for a spot (the picker and the style chips). */
export function ownedFor(decor, spot) {
  return catalogueFor(decor, spot).filter(item => item.owned)
}

/** The shop's status line after buying itemId; room is the write's room (did it land on its spot?). */
export function decorBoughtLine(itemId, room) {
  const entry = decorEntry(itemId)
  if (!entry) return null
  if (entry.spot === ROOM_SPOTS.WALL_STYLE || entry.spot === ROOM_SPOTS.FLOOR_STYLE) return `${entry.label} bought and applied.`
  if (room?.placements?.[entry.spot] === itemId) {

  return `${entry.label} bought. Swap it in with ✏️ Edit room.`
}
```

```js
function squareAround({ x, y, width, height }, pad) {
```

```js
 * server says so (applyRoomWrite). A 409 shows the server's message and refetches the pet so the

 * Station keys and decor ids never collide, so `armed` / `busy` hold either.
 *
 * @param {{ applyRoomWrite: (w: { sprinkles: number, room: Object }) => void, refreshPet: () => void }} input
 * @returns {{ tab: string, setTab: (tab: string) => void, armed: string|null, busy: string|null,
 *   status: string|null, error: string|null, announcement: string|null, arm: (key: string) => void,
 *   upgrade: (station: string) => void, buy: (itemId: string) => void, reset: () => void }}

export function usePetShop({ applyRoomWrite, refreshPet }) {
  const [tab, setTabState] = useState(SHOP_TAB_KEYS.UPGRADES)
  const [armed, setArmed] = useState(null)
  const [busy, setBusy] = useState(null)
  const [status, setStatus] = useState(null)
  const [error, setError] = useState(null)
  const [announcement, setAnnouncement] = useState(null)

  useEffect(() => {
    if (!armed) return undefined
    const timerId = setTimeout(() => setArmed(null), SHOP_CONFIRM_MS)
    return () => clearTimeout(timerId)
  }, [armed])

  useEffect(() => {
    if (!announcement) return undefined
    const timerId = setTimeout(() => setAnnouncement(null), AWARD_BUBBLE_MS)
    return () => clearTimeout(timerId)
  }, [announcement])

  const arm = (key) => {
    setArmed(key)
    setStatus(null)
    setError(null)
  }

  // One purchase at a time; `key` is what shows as busy.
  const purchase = (key, request, onBought, fallbackError) => {
    if (busy) return
    setArmed(null)
    setBusy(key)
    setStatus(null)
    setError(null)
    request()
      .then(result => {
        applyRoomWrite(result)
        onBought(result)
      })
      .catch(err => {
        setError(err.response?.data?.error || fallbackError)
        if (err.response?.status === 409) refreshPet()
      })
      .finally(() => setBusy(null))
  }

  const upgrade = (station) => purchase(station, () => petService.upgradeStation(station), result => {
    setStatus(upgradedLine(station, result.room?.stations?.[station]))
    setAnnouncement(SHOP_UPGRADE_LINES[station] ?? null)
  }, SHOP_UPGRADE_ERROR)

  // MPP-27: the server auto-places decor on an empty spot and applies a style; the line says which.
  const buy = (itemId) => purchase(itemId, () => petService.buyDecor(itemId), result => {
    setStatus(decorBoughtLine(itemId, result.room))
    setAnnouncement(DECOR_BOUGHT_LINE)
  }, DECOR_BUY_ERROR)

  // The sheet closed: forget the armed button and the last message.
  const reset = () => {
    setArmed(null)
    setStatus(null)
    setError(null)
  }

  // A new tab starts clean, like a fresh sheet.
  const setTab = (next) => {
    setTabState(next)
    reset()
  }

  return { tab, setTab, armed, busy, status, error, announcement, arm, upgrade, buy, reset }
}
```

```js
import { useEffect, useRef, useState } from 'react'
import petService from '../services/petService'
import { DECOR_SAVE_ERROR } from '../constants/petDecor'
import { withPlacement } from '../utils/roomPlacements'

/**
 * MPP-27 AC 3: ✏️ Edit room. Placing is free and undoable, so it is optimistic like the wardrobe
 * (MPP-32): the room changes at once (setRoom), the server's RoomWriteDTO replaces it
 * (applyRoomWrite), and a failure rolls back to the last server room with an error line. Only the
 * newest write's outcome is applied, so a slow earlier response can't undo a later tap. Writes are
 * never aborted, and there are no timers. Style chips use place() too (a finish is a spot).
 *
 * @param {{ room: Object|null, setRoom: (room: Object) => void, applyRoomWrite: (w: Object) => void }} input
 * @returns {{ editing: boolean, toggle: () => void, done: () => void, selectedSpot: string|null,
 *   openSpot: (spot: string) => void, closeSpot: () => void,
 *   place: (spot: string, itemId: string|null) => void, error: string|null }}
 */
export function useRoomEdit({ room, setRoom, applyRoomWrite }) {
  const [editing, setEditing] = useState(false)
  const [selectedSpot, setSelectedSpot] = useState(null)
  const [error, setError] = useState(null)
  const seqRef = useRef(0)
  const pendingRef = useRef(0)
  const serverRef = useRef(room)

  // Between writes, the context's copy is the server's.
  useEffect(() => {
    if (pendingRef.current === 0) serverRef.current = room
  }, [room])

  const place = (spot, itemId) => {
    if (!room) return
    const seq = ++seqRef.current
    pendingRef.current += 1
    setError(null)
    setRoom(withPlacement(room, spot, itemId))
    petService.placeDecor(spot, itemId)
      .then(write => {
        serverRef.current = write.room
        if (seq === seqRef.current) applyRoomWrite(write)
      })
      .catch(err => {
        if (seq !== seqRef.current) return
        setRoom(serverRef.current)
        setError(err.response?.data?.error || DECOR_SAVE_ERROR)
      })
      .finally(() => {
        pendingRef.current -= 1
      })
  }

  const done = () => {
    setEditing(false)
    setSelectedSpot(null)
    setError(null)
  }

  return {
    editing,
    toggle: () => (editing ? done() : setEditing(true)),
    done,
    selectedSpot,
    openSpot: setSelectedSpot,
    closeSpot: () => setSelectedSpot(null),
    place,
    error
  }
}

export default useRoomEdit
```

```jsx
function ShopTabs({ tab, idPrefix, onSelect }) {
  const onKeyDown = (event) => {
    const step = event.key === 'ArrowRight' ? 1 : event.key === 'ArrowLeft' ? -1 : 0

    const index = SHOP_TABS.findIndex(t => t.key === tab)
```
